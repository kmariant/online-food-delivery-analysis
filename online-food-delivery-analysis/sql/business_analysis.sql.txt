-- Online Food Delivery Customer Analysis
-- SQL Business Analysis
-- Tool: DuckDB


-- 1. Total number of customers

SELECT COUNT(*) AS total_customers
FROM restaurant_data;


-- 2. Online ordering distribution

SELECT
    "Output",
    COUNT(*) AS customers,
    ROUND(
        COUNT(*) * 100.0 /
        SUM(COUNT(*)) OVER (),
        2
    ) AS percentage
FROM restaurant_data
GROUP BY "Output"
ORDER BY customers DESC;


-- 3. Ordering rate by gender

SELECT
    "Gender",
    COUNT(*) AS customers,
    ROUND(
        SUM(CASE WHEN "Output" = 'Yes' THEN 1 ELSE 0 END)
        * 100.0 / COUNT(*),
        2
    ) AS ordering_rate
FROM restaurant_data
GROUP BY "Gender"
ORDER BY ordering_rate DESC;


-- 4. Ordering rate by occupation

SELECT
    "Occupation",
    COUNT(*) AS customers,
    ROUND(
        SUM(CASE WHEN "Output" = 'Yes' THEN 1 ELSE 0 END)
        * 100.0 / COUNT(*),
        2
    ) AS ordering_rate
FROM restaurant_data
GROUP BY "Occupation"
ORDER BY ordering_rate DESC;


-- 5. Income comparison

SELECT
    "Output",
    COUNT(*) AS customers,
    ROUND(AVG("Monthly Income"), 2) AS average_income,
    ROUND(MEDIAN("Monthly Income"), 2) AS median_income
FROM restaurant_data
GROUP BY "Output";

-- 10. Online customers by occupation

SELECT
    "Occupation",
    COUNT(*) AS online_customers
FROM restaurant_data
WHERE "Output" = 'Yes'
GROUP BY "Occupation"
ORDER BY online_customers DESC;


-- 11. Customer segments

SELECT
    "Gender",
    "Occupation",
    "Customer Type",
    COUNT(*) AS customers,
    SUM(
        CASE WHEN "Output" = 'Yes' THEN 1 ELSE 0 END
    ) AS online_customers
FROM restaurant_data
GROUP BY
    "Gender",
    "Occupation",
    "Customer Type"
ORDER BY online_customers DESC;


-- 12. Top occupations by ordering rate

SELECT
    "Occupation",
    COUNT(*) AS customers,
    ROUND(
        SUM(CASE WHEN "Output" = 'Yes' THEN 1 ELSE 0 END)
        * 100.0 / COUNT(*),
        2
    ) AS ordering_rate
FROM restaurant_data
GROUP BY "Occupation"
HAVING COUNT(*) >= 5
ORDER BY ordering_rate DESC;
